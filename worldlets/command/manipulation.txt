!command/get!
parent = "generic/char_command"
category = "Object Manipulation"
name = "get"
syntax = "(#quantity#) <object>"
help = """
Get an object.

Syntax:
    get <object name>
    get <quantity> <object name>

Get one or several objects from the current room (character location)
to the character's inventory (character's contents). You can specify
a number of objects to get, like get 3 apples.
"""

def refine(character, object, quantity=1):
    to_pick = search.match(character.location, object, viewer=character, limit=quantity)

    if not to_pick:
        character.msg("You don't see that around.")
        raise RefineError
    endif

def run(character, to_pick):
    for object in to_pick:
        object.location = character
    done

    for name in names.group(to_pick, viewer=character):
        character.msg(f"You pick up {name}.")
    done

!command/drop!
parent = "generic/char_command"
category = "Object manipulation"
name = "drop"
syntax = "(#quantity#) <object>"
help = """
Drop one or several objects.

Syntax:
    drop <object name>
    drop <quantity> <object name>

Drop one or several objects from the inventory (characters' contents)
to the room (character location). You can specify the object name
or a quantity followed by the object name.
"""

def refine(character, object, quantity=1):
    to_drop = search.match(character, object, viewer=character, limit=quantity)

    if not to_drop:
        character.msg("You don't see that around.")
        raise RefineError
    endif

def run(character, to_drop):
    for object in to_drop:
        object.location = character.location
    done

    for name in names.group(to_drop, viewer=character):
        character.msg(f"You drop {name} on the ground.")
    done
