def __init__(self, text: str):
    self.text = text

def format_for(self, location: Entity, viewer: Entity):
    """Format the text description in proper paragraphs."""
    lines = []
    for paragraph in self.text.split("\n\n"):
        formatted = self.build_dynamic(paragraph, location, viewer)
        lines.extend(display.wrap(formatted, width=75, initial_indent='   '))
    done

    return "\n".join(lines)

def build_dynamic(paragraph: str, location: Entity, viewer: Entity):
    parts = []

    for begin, _before, name, end in paragraph.scan_between("$", ",.?!- \n"):
        parts.append(begin)

        if name == "$":
            parts.append(name)
        else:
            method = getattr(location, f"describe_{name}", None)

            if method is not None:
                text = method(viewer)
                parts.append(text)
            endif
        endif

        parts.append(end)
    done

    if len(parts) == 0:
        return paragraph
    else:
        return "".join(parts)
    endif

        text = "- " + "\n- ".join(grouped)

    # Display objects

        lines.append("")

def look_exits(self, character: Entity):
    """Return a message showing the available exits for a given character."""
    exits = []
    is_admin = "admin" in character.permissions

    for exit in self.exits:
        if exit.hidden:
            if is_admin:
                exits.append(f"{exit.name}(i)")
            endif
        else:
            exits.append(exit.name)
        endif
    done

    if exits:
        return "Obvious exits: " + ", ".join(exits) + "."
    else:
        return "There is no obvious exit here."
    endif

def find_exit(self, name: str, allow_partial: bool = False):
    """Find an exit depending on a name."""
    exits = self.exits

    for exit in exits:
        if exit.match_name(name, allow_partial=allow_partial):
            return exit
        endif
    done

    return None

!Exit!
name = None
destination = None

def __init__(self, name: str, destination: Entity, aliases = None, hidden: bool = False):
    self.name = name
    self.destination = destination
    self.aliases = aliases
    self.hidden = hidden

def match_name(self, name: str, allow_partial: bool = False):
    """Check if the exit's name or aliases match the given name."""
    name = name.lower()
    aliases = self.aliases

    if aliases is None:
        aliases = []
    endif

    if name == self.name.lower() or name in aliases:
        return True
    elif not self.hidden and allow_partial:
        if self.name.lower().startswith(name):
            return True
        endif

        for alias in aliases:
            if alias.lower().startswith(name):
                return True
            endif
        done
    endif

    return False

!generic/channel!
name = None
aliases = []
outside = set()
permissions = set()

def join(self, person: Entity):
    self.outside.discard(person)

def leave(self, person: Entity):
    self.outside.add(person)

def present(self, person: Entity):
    return person not in self.outside

def can_listen(self, person: Entity):
    """Return True if this person can listen to this channel, False otherwise."""
    for permission in self.permissions:
        if permission not in person.permissions:
            return False
        endif
    done

    return True

def broadcast(self, person: Entity, message: str):
    """Broadcast a message to all members of the channel."""
    to_sender = self.format_for_sender(person, message)
    to_others = self.format_for_others(person, message)

    # Send the message to the sender.
    person.msg(to_sender)

    # Browse the list of clients and build the list of others.
    others = set()
    for client in clients.active():
        owner = getattr(client, "owner", None)
        if owner is not None and owner is not person:
            if self.present(owner) and self.can_listen(owner):
                others.add(owner)
            endif
        endif
    done

    # Send the message to all other entities.
    for other in others:
        other.msg(to_others)
    done

def format_for_sender(self, person: Entity, message: str):
    """Return the message to send to the sender."""
    return f"[{self.name}] You say: {message}"

def format_for_others(self, person: Entity, message: str):
    """Return the message to send to other players."""
    return f"[{self.name}] {person.name} says: {message}"

def match_name(self, person: Entity, message: str):
    """Returns whether the message has been broadcasted to any dhannel."""
    if " " in message:
        name, message = message.split(" ", 1)
    else:
        name = message
        message = None
    endif

    name = name.lower()

    # Look for a matching channel.
    for channel in self.children:
        if channel.name == name or name in channel.aliases:
            if message:
                channel.broadcast(person, message)
            else:
                person.msg("What do you want to say on this channel?")
            endif
            return True
        endif
    done

    return False
