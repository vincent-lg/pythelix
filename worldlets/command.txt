!command/help!
parent = "generic/char_command"
name = "help"
category = "Information"

def run(character):
    commands = !menu/game!.get_commands(character)
    categories = {
        "General": [],
        "Information": [],
        "Administrator": [],
    }

    for command in commands:
        category = command.category
        if category in categories:
            categories[category].append(command.name)
        endif
    done

    hlist = display.HorizontalList(columns=4, col_width=15)
    for category in categories.keys():
        commands = categories[category]
        if commands:
            commands.sort()
            hgroup = hlist.add_group(category)
            for command in commands:
                hgroup.add_entry(command)
            done
        endif
    done

    character.msg(hlist.format())

!command/quit!
parent = "generic/player_command"
name = "quit"
syntax = ""
category = "General"

def run(character):
    character.former_location = character.location
    character.location = None
    client = clients.owning(character)
    client.msg("Good bye!")
    client.disconnect()

!command/get!
parent = "generic/char_command"
name = "get"
category = "General"
syntax = "(#quantity#) <object>"

def refine(character, object, quantity=1):
    to_pick = search.match(character.location, object, viewer=character, limit=quantity)

    if not to_pick:
        character.msg("You don't see that around.")
        raise RefineError
    endif

def run(character, to_pick):
    for object in to_pick:
        object.location = character
    done

    for name in names.group(to_pick, viewer=character):
        character.msg(f"You pick up {name}.")
    done
