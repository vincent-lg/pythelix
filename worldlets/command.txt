!command/help!
parent = "generic/char_command"
name = "help"
category = "Information"

def run(character):
    commands = !menu/game!.get_commands(character)
    categories = {
        "General": [],
        "Information": [],
        "Administrator": [],
    }

    for command in commands:
        category = command.category
        if category in categories:
            categories[category].append(command.name)
        endif
    done

    hlist = display.HorizontalList(columns=4, col_width=15)
    for category in categories.keys():
        commands = categories[category]
        if commands:
            hgroup = hlist.add_group(category)
            for command in commands:
                hgroup.add_entry(command)
            done
        endif
    done

    character.msg(hlist.format())

def run(character):
    character.former_location = character.location
    character.location = None
    client = clients.owning(character)
    client.msg("Good bye!")
    client.disconnect()

!command/eye!
parent = "generic/char_command"
category = "General"
name = "eye"

def run(character):
    colors = ("blue", "green", "brown", "orange", "black", "white")
    choices = {}
    prompt = "Enter the eye color as a number:"
    i = 1

    for color in colors:
        choices[str(i)] = color
        prompt += f"\n  {i} for {color}"
        i += 1
    done

    prompt += f"\nEnter a number between 1 and {len(colors)}:"
    retry = "Enter a valid number."
    color = choice(character, choices, prompt=prompt, retry=retry)
    character.msg(f"You have chosen the eye color {color}.")
