def input(client, input):
    username = input.strip().lower()
    account = search.one(!generic/account!, username=username)

    if len(username) < 3:
        client.msg("This username isn't valid. Choose another one.")
    elif account:
        client.msg("This account already exists. Try a new one.")
    else:
        account = Entity(parent=!generic/account!)
        account.username = username
        client.account = account
        client.location = !menu/new/password!
    endif

def unknown_input(client, input):
    account = client.account
    has_players = len(!generic/player!.children) > 0
    player = Entity(parent=!generic/player!)
    player.name = input
    player.account = account
    account.players.append(player)

    if not has_players:
        log(f"{player.name!r} (#{player.id}) becomes an administrator")
        player.permissions.add("admin")
    endif

    player.former_location = !room/bakery!
    client.player = player
    client.location = !menu/near_game!
