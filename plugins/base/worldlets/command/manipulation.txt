!command/get!
parent = "generic/char_command"
category = "Object Manipulation"
name = "get"
syntax = "(#quantity#) <object>"
help = """
Get an object.

Syntax:
    get <object name>
    get <quantity> <object name>

Get one or several objects from the current room (character location)
to the character's inventory (character's contents). You can specify
a number of objects to get, like get 3 apples.
"""

def refine(character, object, quantity=1):
    to_pick = character.match(object, limit=quantity)

    if not to_pick:
        character.msg("You don't see that around.")
        raise RefineError
    endif

def run(character, to_pick):
    for object in to_pick:
        object.location = character
    done

    for name in names.group(to_pick, viewer=character):
        character.msg(f"You pick up {name}.")
    done

!command/drop!
parent = "generic/char_command"
category = "Object manipulation"
name = "drop"
syntax = "(#quantity#) <object>"
help = """
Drop one or several objects.

Syntax:
    drop <object name>
    drop <quantity> <object name>

Drop one or several objects from the inventory (characters' contents)
to the room (character location). You can specify the object name
or a quantity followed by the object name.
"""

def refine(character, object, quantity=1):
    to_drop = character.match(object, candidates=character.contents, limit=quantity)

    if not to_drop:
        character.msg("You don't see that around.")
        raise RefineError
    endif

def run(character, to_drop):
    for object in to_drop:
        object.location = character.location
    done

    for name in names.group(to_drop, viewer=character):
        character.msg(f"You drop {name} on the ground.")
    done

!command/give!
parent = "generic/char_command"
category = "Object manipulation"
name = "give"
syntax = "(#quantity#) <object> to <person>"
help = """
Give one or several objects to someone else.

Syntax:
    give <object name> to <character name>
    give <quantity> <object name> to <character name>

Give one or more objects to someone else in the room. The given person
must be a character present in the same room.
"""

def refine(character, object, person, quantity=1):
    to_give = character.match(object, candidates=character.contents, limit=quantity)

    if not to_give:
        character.msg("You don't see that around.")
        raise RefineError
    endif

    dest = character.match(person, limit=quantity)

    if not dest:
        character.msg("You don't see this person around.")
        raise RefineError
    endif

    person = dest[0]

    if not isinstance(person, !generic/character!):
        character.msg(f"You cannot give anything to {person}.")
        raise RefineError
    endif

def run(character, to_give, person):
    for object in to_give:
        object.location = person
    done

    for name in names.group(to_give, viewer=character):
        character.msg(f"You give {name} to {person}.")
        person.msg(f"{character} gives you {name}.")
        character.location.announce(f"{character} gives {name} to {person}.")
    done
