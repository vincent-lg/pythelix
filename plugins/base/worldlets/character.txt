def msg(self, text, prompt: bool = True):

def try_exit(exit):
    """Move the character through an exit, or try to."""
    if not self.location.can_leave(self, exit):
        return False
    endif

    self.location.leave(self, exit)
    self.location.announce(f"{self} leaves {exit.name}.")
    self.location = exit.destination
    self.msg(self.location.look(self))
    self.location.announce(f"{self} arrives.")
    self.location.enter(self)

    return True
