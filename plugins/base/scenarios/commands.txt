# The commands every game starts with.

given Alice in !base/room/start!
given Bob in !base/room/start!

Alice> look

Alice> say is anyone here

Bob sees "is anyone here"

Bob> help
Bob sees "look"
Bob sees "say"

Bob> xyzzy
Bob sees "Command not found"
Alice does not see "Command not found"
